function initAuth() { const loginForm = document.getElementById("loginForm"); if (loginForm) { loginForm.addEventListener("submit", function(event) { event.preventDefault(); const phone = document.getElementById("phone")?.value.trim(); const password = document.getElementById("password")?.value; if (!phone || !password) { const msg = document.getElementById("msg"); if (msg) msg.textContent = "Please enter your phone number and password."; return; } localStorage.setItem( "earnlinkUser", JSON.stringify({ phone: phone }) ); window.location.href = "index.html"; }); } const registerForm = document.getElementById("registerForm"); if (registerForm) { registerForm.addEventListener("submit", function(event) { event.preventDefault(); const phone = document.getElementById("phone")?.value.trim(); const password = document.getElementById("password")?.value; if (!phone || !password) { const msg = document.getElementById("msg"); if (msg) msg.textContent = "Please fill in all required fields."; return; } localStorage.setItem( "earnlinkUser", JSON.stringify({ phone: phone }) ); window.location.href = "index.html"; }); } }
